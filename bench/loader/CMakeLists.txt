cmake_minimum_required(VERSION 3.16)
# Also usable directly: no AdaptiveCpp, SQL parser or application dependencies.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(crystal-ssb-loader LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
endif()
get_filename_component(CRYSTAL_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT TARGET checked_sizes)
    add_library(checked_sizes STATIC "${CRYSTAL_SOURCE_ROOT}/src/core/size_expression.cpp")
    target_include_directories(checked_sizes PUBLIC "${CRYSTAL_SOURCE_ROOT}/include")
endif()
add_library(ssb_storage STATIC
    "${CRYSTAL_SOURCE_ROOT}/src/storage/ssb_dataset.cpp"
    "${CRYSTAL_SOURCE_ROOT}/src/storage/text_column.cpp")
target_include_directories(ssb_storage PUBLIC "${CRYSTAL_SOURCE_ROOT}/include")
target_link_libraries(ssb_storage PUBLIC checked_sizes)
add_library(ssb_importer STATIC "${CRYSTAL_SOURCE_ROOT}/src/storage/ssb_import.cpp")
target_link_libraries(ssb_importer PUBLIC ssb_storage)
add_executable(ssb_convert ssb_convert.cpp)
target_link_libraries(ssb_convert PRIVATE ssb_importer)
add_executable(test_ssb_dataset "${CRYSTAL_SOURCE_ROOT}/test/ssb_native_data/test_dataset.cpp" "${CRYSTAL_SOURCE_ROOT}/src/core/config.cpp")
target_include_directories(test_ssb_dataset PRIVATE "${CRYSTAL_SOURCE_ROOT}")
target_compile_options(test_ssb_dataset PRIVATE -UNDEBUG)
target_link_libraries(test_ssb_dataset PRIVATE ssb_importer)
enable_testing()
add_test(NAME test_ssb_dataset COMMAND test_ssb_dataset)
add_executable(ssb_reference "${CRYSTAL_SOURCE_ROOT}/test/ssb_native_data/oracle.cpp")
