cmake_minimum_required(VERSION 3.20)
project(pcuda_stage8_analysis LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
set(PROJECT_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../include")
add_executable(test_probe_log test_probe_log.cpp probe_log.cpp)
target_include_directories(test_probe_log PRIVATE "${PROJECT_INCLUDE}")
target_link_libraries(test_probe_log PRIVATE Threads::Threads)
enable_testing()
add_test(NAME probe_log COMMAND test_probe_log)
add_library(probe_fixture SHARED probe_fixture.cpp)
add_executable(test_probe_symbols test_probe_symbols.cpp)
target_link_libraries(test_probe_symbols PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME probe_local_symbols COMMAND test_probe_symbols "$<TARGET_FILE:probe_fixture>")
add_executable(test_probe_preload test_probe_preload.cpp)
target_link_libraries(test_probe_preload PRIVATE ${CMAKE_DL_LIBS})

option(BUILD_S2_PROBE "Build the version-bound diagnostic ELF interposer" ON)
if(BUILD_S2_PROBE)
  find_package(LLVM 17 REQUIRED CONFIG)
  set(ACPP_PREFIX "" CACHE PATH "Installed AdaptiveCpp prefix")
  set(ROCM_PREFIX "" CACHE PATH "Installed HIP SDK prefix")
  find_path(ACPP_INCLUDE hipSYCL/compiler/llvm-to-backend/LLVMToBackend.hpp HINTS "${ACPP_PREFIX}/include/AdaptiveCpp" REQUIRED)
  find_library(ACPP_BACKEND llvm-to-backend HINTS "${ACPP_PREFIX}/lib/hipSYCL/llvm-to-backend" REQUIRED)
  find_library(LLVM_SHARED NAMES LLVM-17git LLVM-17 LLVM HINTS ${LLVM_LIBRARY_DIRS} REQUIRED)
  find_path(HIP_INCLUDE hip/hiprtc.h HINTS "${ROCM_PREFIX}/include" REQUIRED)
  find_library(HIP_RUNTIME amdhip64 HINTS "${ROCM_PREFIX}/lib" REQUIRED)
  if(NOT LLVM_VERSION_MAJOR EQUAL 17 OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "S2 probe is a research-only Linux/LLVM17 interposer")
  endif()
  add_library(s2_probe SHARED probe_log.cpp probe_jit.cpp probe_runtime.cpp)
  target_include_directories(s2_probe PRIVATE "${PROJECT_INCLUDE}")
  target_include_directories(s2_probe SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} "${ACPP_INCLUDE}" "${HIP_INCLUDE}")
  target_compile_options(s2_probe PRIVATE -fno-rtti)
  target_compile_definitions(s2_probe PRIVATE CRYSTAL_HIP_RUNTIME="${HIP_RUNTIME}")
  target_link_libraries(s2_probe PRIVATE "${ACPP_BACKEND}" "${LLVM_SHARED}" ${CMAKE_DL_LIBS} Threads::Threads)
endif()
