cmake_minimum_required(VERSION 3.20)
project(pcuda_jit_tiering LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LLVM 17 REQUIRED CONFIG)
set(ACPP_PREFIX "" CACHE PATH "Installed AdaptiveCpp prefix")
find_path(ACPP_INCLUDE hipSYCL/compiler/llvm-to-backend/LLVMToBackend.hpp
    HINTS "${ACPP_PREFIX}/include/AdaptiveCpp" REQUIRED)
find_library(ACPP_BACKEND llvm-to-backend HINTS "${ACPP_PREFIX}/lib/hipSYCL/llvm-to-backend" REQUIRED)
find_library(ACPP_AMDGPU llvm-to-amdgpu HINTS "${ACPP_PREFIX}/lib/hipSYCL/llvm-to-backend" REQUIRED)
find_library(LLVM_SHARED NAMES LLVM-17git LLVM-17 LLVM HINTS ${LLVM_LIBRARY_DIRS} REQUIRED)
set(ROCM_PREFIX "" CACHE PATH "HIP SDK prefix for the instrumented AMD backend")
find_path(HIP_INCLUDE hip/hiprtc.h HINTS "${ROCM_PREFIX}/include" REQUIRED)
if(NOT LLVM_VERSION_MAJOR EQUAL 17 OR NOT UNIX OR APPLE)
    message(FATAL_ERROR "This experiment is version-bound to LLVM 17 on Linux")
endif()
add_library(jit_control SHARED jit_control.cpp)
target_include_directories(jit_control SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} "${ACPP_INCLUDE}" "${HIP_INCLUDE}")
target_compile_options(jit_control PRIVATE -fno-rtti)
target_link_libraries(jit_control PRIVATE "${ACPP_AMDGPU}" "${ACPP_BACKEND}" "${LLVM_SHARED}" ${CMAKE_DL_LIBS})
